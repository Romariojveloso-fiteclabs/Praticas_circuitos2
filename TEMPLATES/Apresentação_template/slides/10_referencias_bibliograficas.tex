}

  \ifitemBibliograficoAberto
    \end{minipage}%
  \fi
  \itemBibliograficoOriginal{#1}%
  \begin{minipage}[t]{0.97\linewidth}%
  \setlength{\parskip}{0pt}%
  \itemBibliograficoAbertotrue
}
\renewcommand{\endthebibliography}{%
  \ifitemBibliograficoAberto
    \end{minipage}%
  \fi
  \fimBibliografiaOriginal
}
